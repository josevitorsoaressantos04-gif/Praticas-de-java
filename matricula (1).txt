import java.util.Scanner;


public class Main {
    public static void main(String[] args) throws InterruptedException {
        Scanner sc = new Scanner(System.in);

        String nomeCurso = "", nomeAluno = "", rs = "";
        int idade = 0, opcao = 0, idadeMin = 0;
        double cgH = 0, cghAluno = 0;

        System.out.println("""
                === verificação requesitos para matricula ===
                
                1 - primeiro digite seu nome:""");
        nomeAluno = sc.nextLine();
        System.out.println("""
                2 - Digite a sua idade:""");
        idade = sc.nextInt();
        sc.nextLine();
        do {

            System.out.println("""
                    === Opções De Cursos===
                    
                    código         nome             carga horária      idade miníma
                    1        Matemática Básica       40 horas             nenhum
                    2        Pogramação Java         60 horas             14 anos
                    3        Física Experimental     50 horas             15 anos
                    4        Inglês Intermediário    45 horas             12 anos
                    5        Robótica                70 horas             16 anos
                    
                    Digite a opção do curso que deseja:
                    """);
            opcao = sc.nextInt();
            sc.nextLine();

            switch (opcao) {
                case 1:
                    nomeCurso = "Matemática Básica";
                    idadeMin = 0;

                    cgH = 40;
                    if (idade < idadeMin) {
                        System.out.printf("%nInfelizmente %s, você não tem a idade mínima para a matricula", nomeAluno);
                    } else {
                        System.out.printf("%nMatriculá em %s Realizada com sucesso.", nomeCurso);
                        cghAluno += cgH;
                    }
                    System.out.println();
                    break;

                case 2:
                    nomeCurso = "Pogramação Java ";
                    idadeMin = 14;
                    cgH = 60;
                    if (idade < idadeMin) {
                        System.out.printf("%nInfelizmente %s, você não tem a idade mínima para a matricula%n", nomeAluno);
                    } else {
                        System.out.printf("%nMatriculá em %s Realizada com sucesso.%n", nomeCurso);
                        cghAluno += cgH;

                    }

                    break;

                case 3:
                    nomeCurso = "Física Experimental";
                    idadeMin = 15;

                    if (idade < idadeMin) {
                        System.out.printf("%nInfelizmente %s, você não tem a idade mínima para a matricula%n", nomeAluno);
                    } else {
                        cgH = 50;
                        System.out.printf("%nMatriculá em %s Realizada com sucesso.%n", nomeCurso);
                        cghAluno += cgH;
                    }
                    break;

                case 4:
                    nomeCurso = "Inglês Intermediário%n";
                    idadeMin = 12;

                    if (idade < idadeMin) {
                        System.out.printf("%nInfelizmente %s, você não tem a idade mínima para a matricula%n", nomeAluno);
                    } else {
                        cgH = 45;
                        System.out.printf("%nMatriculá em %s Realizada com sucesso.%n", nomeCurso);
                        cghAluno += cgH;
                    }
                    break;

                case 5:
                    nomeCurso = "Robótica";
                    idadeMin = 16;
                    cgH = 70;
                    if (idade < idadeMin) {
                        System.out.printf("%nInfelizmente %s, você não tem a idade mínima para a matricula%n", nomeAluno);
                    } else {
                        System.out.printf("%nMatriculá em %s Realizada com sucesso %n.", nomeCurso);
                        cghAluno += cgH;
                    }
                    break;

            }
            System.out.println("Deseja se matricular em mais algum curso(s/n)?: ");
            rs = sc.nextLine();
        } while (opcao != 0 && !rs.equals("n"));

        if(cghAluno > 0 ){
        System.out.println("Gerando comprovante...");
        for(int i = 1; i < 4; i++){
            System.out.println(i);
            Thread.sleep(1000);
        }
        System.out.printf("""
                === Comprovante  Matricula ===
                
                Nome do aluno: %s
                Idade do aluno: %d
                Carga de Horas Total: %.2f
                
                """, nomeAluno, idade,cghAluno);
        }
        else{
            System.out.println("Você não se matriculou em nenhum curso");
        }

        sc.close();
    }
}
