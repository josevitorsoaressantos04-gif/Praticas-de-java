import java.util.Scanner;

public class Atividade4 {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);

        double totalVendas, pcComissao,pagamento;
        pcComissao = 5 ;// porcentagem da comissao

        System.out.print("Digite qual foi o total de venda esse mês:");
        totalVendas = scan.nextDouble();

        pagamento = totalVendas * (pcComissao/100);

        System.out.printf("Sua comissão é de: R$ %.2f", pagamento);




    }
}

import java.util.Scanner;

public class Atividade5 {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);

        double produto, valorTotal;
        int parcelas;

        System.out.print("Digite o valor do produto: ");
        produto = scan.nextDouble();

        System.out.print("Vai parcelar em quantas vezes?");
        parcelas = scan.nextInt();


        valorTotal = produto / parcelas;

        System.out.printf("O valor do produto é: %.2f%n" +
                          "Parcelado em %dXaa de R$ %.2f",
                          produto, parcelas, valorTotal
        );

    }
}
