import java.util.Scanner;

public class Atividade5 {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);

        double produto, valorTotal;
        int parcelas;

        System.out.print("Digite o valor do produto: ");
        produto = scan.nextDouble();

        System.out.print("Vai parcelar em quantas vezes?");
        parcelas = scan.nextInt();


        valorTotal = produto / parcelas;

        System.out.printf("O valor do produto é: %.2f%n" +
                          "Parcelado em %dXaa de R$ %.2f",
                          produto, parcelas, valorTotal
        );

    }
}
